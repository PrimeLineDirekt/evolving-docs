#### Example



**Code:**
```bash
# Verify no orphan edges
for edge in $(jq -r '.[].target' _graph/edges.json); do
  grep -q "\"id\": \"$edge\"" _graph/*-nodes.json || echo "ORPHAN: $edge"
done

# Verify indexes are current
node_count=$(jq 'length' _graph/core-nodes.json)
index_count=$(jq '[.[]] | add | length' _graph/index/core-by-type.json)
[ "$node_count" -eq "$index_count" ] || echo "INDEX STALE!"
```
